import hashlib,json,subprocess
from examples.huggingface.benchmark import measure
import argparse
p=argparse.ArgumentParser();p.add_argument('--samples',type=int,default=100);p.add_argument('--warmup',type=int,default=10);p.add_argument('--output',required=True);a=p.parse_args()
r=measure('pytorch',real=True,samples=a.samples,warmup=a.warmup,batch_size=1,sequence_length=4096,chunk_size=128,nf4=True,projection_precision='fp32',checkpoint_layers=4,train_file='/tmp/onnax-qwen-longalpaca-4k-with-benchmark/benchmark.jsonl',progress_every=1,torch_device='mps',torch_chunked=True,torch_attention='sdpa',torch_mps_memory_limit_bytes=20*1024**3,validate_updates=True)
r['source_commit']=subprocess.check_output(['git','rev-parse','HEAD'],text=True).strip()

json.dump(r,open(a.output,'w'),indent=2)
print(json.dumps(r['timings_ms']['step']),flush=True)
