"""Run paired measurements after the identified regression process succeeds."""
import hashlib, json, os, subprocess, sys, time, xml.etree.ElementTree as ET
from pathlib import Path
root=Path('/tmp/onnax-symbolic-formal-final');root.mkdir(exist_ok=False)
def status(phase,**extra):
    (root/'status.json').write_text(json.dumps({'phase':phase,**extra},indent=2)+'\n')
try:
    status('waiting_for_regression',pid=52358)
    while True:
        try: os.kill(52358,0)
        except ProcessLookupError: break
        time.sleep(2)
    xml=ET.parse('/tmp/onnax-symbolic-adjoint-full-final.xml')
    cases=list(xml.iter('testcase'))
    assert len(cases)>3900, len(cases)
    assert not list(xml.iter('failure')) and not list(xml.iter('error')), 'Regression did not pass'
    files=[Path('src/onnax/_autodiff/_context.py'),Path('/tmp/onnax-symbolic-native-worker.py'),Path('/tmp/onnax-symbolic-full-qualification.py')]
    frozen={str(p):hashlib.sha256(p.read_bytes()).hexdigest() for p in files}
    (root/'sources.json').write_text(json.dumps(frozen,indent=2)+'\n')
    for seq in [64,4032]:
        status('measuring',sequence=seq)
        env={**os.environ,'ONNAX_CHECK_SEQUENCE':str(seq),'ONNAX_CHECK_WARMUP':'10','ONNAX_CHECK_SAMPLES':'100','ONNAX_CHECK_OUTPUT':str(root/f'sequence{seq}')}
        with (root/f'sequence{seq}.log').open('w') as log:
            subprocess.run([sys.executable,'/tmp/onnax-symbolic-full-qualification.py'],env=env,stdout=log,stderr=subprocess.STDOUT,check=True)
        for p in files: assert hashlib.sha256(p.read_bytes()).hexdigest()==frozen[str(p)], str(p)
    status('complete')
except BaseException as error:
    status('failed',error=repr(error));raise
