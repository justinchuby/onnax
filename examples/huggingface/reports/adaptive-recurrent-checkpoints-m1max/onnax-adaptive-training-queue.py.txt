import json,os,subprocess,time
from pathlib import Path
import numpy as np
root=Path('/tmp/onnax-adaptive-context-training');root.mkdir(exist_ok=True)
python='/Users/justinc/Documents/GitHub/onnax/.venv/bin/python';cwd='/tmp/onnax-adaptive-recurrent-checkpoints'
state={'phase':'waiting_for_adaptive_gpu_cases'}
def save():
 p=root/'status.json';q=p.with_suffix('.tmp');q.write_text(json.dumps(state,indent=2)+'\n');q.replace(p)
save();deadline=time.monotonic()+10800
try:
 while True:
  d=json.loads(Path('/tmp/onnax-adaptive-state-measurement/status.json').read_text())
  assert d['phase']!='stopped',d
  if d['phase']=='complete':break
  assert time.monotonic()<deadline,'GPU wait deadline'
  time.sleep(15)
 env={**os.environ,'PYTHONPATH':'src:.','HF_HUB_OFFLINE':'1','HF_DATASETS_OFFLINE':'1'}
 for name in ['continuous','resumed']:
  state.update(phase=name);save();command=[python,'/tmp/onnax-adaptive-context-training.py','--output',str(root/name)]
  if name=='resumed':command+=['--resume',str(root/'continuous')]
  with open(root/(name+'.log'),'w') as log:
   subprocess.run([python,'-m','examples.huggingface.guard','--output-dir',str(root/(name+'-guard')),'--',*command],cwd=cwd,env=env,stdout=log,stderr=subprocess.STDOUT,check=True)
 a=np.load(root/'continuous/final.npz');b=np.load(root/'resumed/final.npz');assert set(a.files)==set(b.files)
 for name in a.files:assert np.array_equal(a[name],b[name]),name
 state.update(phase='complete',fresh_process_resume_bitwise_arrays=len(a.files));save()
except Exception as exc:state.update(phase='stopped',error=repr(exc));save();raise
print(json.dumps(state),flush=True)
